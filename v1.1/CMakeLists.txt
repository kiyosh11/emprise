cmake_minimum_required(VERSION 3.24)
project(emprise VERSION 1.1.0 LANGUAGES C CXX)
include(CTest)
find_package(OpenMP QUIET)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/pcre2-10.45 EXCLUDE_FROM_ALL)
add_library(utf8proc STATIC third_party/utf8proc/utf8proc.c)
target_include_directories(utf8proc PUBLIC third_party/utf8proc)
target_compile_definitions(utf8proc PUBLIC UTF8PROC_STATIC)
add_library(emprise src/gguf.cpp src/quant.cpp src/tokenizer.cpp src/backend.cpp src/cuda.cpp src/model.cpp src/hardware.cpp)
target_compile_features(emprise PUBLIC cxx_std_20)
target_include_directories(emprise PUBLIC include PRIVATE third_party)
target_link_libraries(emprise PRIVATE pcre2-8-static utf8proc ${CMAKE_DL_LIBS})
target_compile_definitions(emprise PRIVATE PCRE2_STATIC PCRE2_CODE_UNIT_WIDTH=8)
if(OpenMP_CXX_FOUND)
  target_link_libraries(emprise PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(emprise PRIVATE /W4 /permissive-)
else()
  target_compile_options(emprise PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(emprise-inspect src/inspect.cpp)
target_link_libraries(emprise-inspect PRIVATE emprise)
add_executable(emprise-chat src/chat.cpp)
target_link_libraries(emprise-chat PRIVATE emprise)
target_include_directories(emprise-chat PRIVATE third_party)
if(OpenMP_CXX_FOUND)
  target_link_libraries(emprise-chat PRIVATE OpenMP::OpenMP_CXX)
endif()
add_executable(emprise-fused-check tests/fused_check.cpp)
target_link_libraries(emprise-fused-check PRIVATE emprise)
target_include_directories(emprise-fused-check PRIVATE third_party)
add_executable(emprise-kernel-bench tests/kernel_bench.cpp)
target_link_libraries(emprise-kernel-bench PRIVATE emprise)
add_executable(emprise-block-check tests/block_check.cpp)
target_link_libraries(emprise-block-check PRIVATE emprise)
target_include_directories(emprise-block-check PRIVATE third_party)
add_executable(emprise-mtp-check tests/mtp_check.cpp)
target_link_libraries(emprise-mtp-check PRIVATE emprise)
target_include_directories(emprise-mtp-check PRIVATE third_party)
if(BUILD_TESTING)
  add_executable(emprise-tests tests/core.cpp)
  target_link_libraries(emprise-tests PRIVATE emprise)
  add_test(NAME emprise-core COMMAND emprise-tests)
endif()
